#+begin_src emacs-lisp
(use-package yank-media
  :ensure t
  :bind (("C-c v" . yank-media))
  :config
  (setq yank-media-create-directory t)
  (setq yank-media-image-directory "assets/images")
  ;; Auto-select image/png when available
  (defun my/yank-media-auto-png (orig-fun &rest args)
    "Automatically select image/png when available."
    (cl-letf (((symbol-function 'completing-read)
               (lambda (prompt collection &optional predicate require-match
                               initial-input hist def inherit-input-method)
                 (let ((options (all-completions "" collection predicate)))
                   (if (member "image/png" options)
                       "image/png"
                     (completing-read-default prompt collection predicate require-match
                                              initial-input hist def inherit-input-method))))))
      (apply orig-fun args)))
  (advice-add 'yank-media :around #'my/yank-media-auto-png))

(defun my/org-yank-image-handler (_type image)
  "Handle pasting images in org-mode."
  (let ((dir (expand-file-name yank-media-image-directory default-directory)))
    (unless (file-directory-p dir)
      (make-directory dir t))
    (let ((file (make-temp-file
                 (expand-file-name (format-time-string "%Y%m%d_%H%M%S_") dir)
                 nil ".png")))
      (with-temp-file file
        (set-buffer-multibyte nil)
        (insert image))
      (insert (format "[[file:%s]]" (file-relative-name file)))
      (org-display-inline-images))))

(with-eval-after-load 'org
  (setq org-startup-with-inline-images t)
  (setq org-image-actual-width 1000)
  (define-key org-mode-map (kbd "C-c v") #'yank-media)
  (add-hook 'org-mode-hook
            (lambda ()
              (yank-media-handler "image/.*" #'my/org-yank-image-handler))))

  (defun my/rename-file-and-buffer (new-name)
    "Rename the current file and its buffer to NEW-NAME without overwriting."
    (interactive
     (progn
	(unless buffer-file-name
	  (user-error "Current buffer is not visiting a file"))
	(list (read-file-name "Rename file to: "
			      (file-name-directory buffer-file-name)
			      nil nil
			      (file-name-nondirectory buffer-file-name)))))
    (unless buffer-file-name
      (user-error "Current buffer is not visiting a file"))
    (let ((new-name (expand-file-name new-name)))
      (when (file-equal-p buffer-file-name new-name)
	(user-error "New name is the same as the current file"))
      (when (get-file-buffer new-name)
	(user-error "A buffer is already visiting %s" new-name))
      (rename-file buffer-file-name new-name nil)
      (set-visited-file-name new-name t)
      (set-buffer-modified-p nil)
      (message "Renamed file to %s" new-name)))

  (defun my/move-buffer-file (dir)
    "Moves both current buffer and file it's visiting to DIR."
    (interactive "DNew directory: ")
    (if-let* ((filename (buffer-file-name))
	      (newname (expand-file-name (file-name-nondirectory filename) dir)))
	(progn
	  (when (file-exists-p newname)
	    (user-error "Destination already exists: %s" newname))
	  (when (get-file-buffer newname)
	    (user-error "A buffer is already visiting %s" newname))
	  (rename-file filename newname nil)
	  (set-visited-file-name newname t)
	  (set-buffer-modified-p nil))
      (user-error "Current buffer is not visiting a file")))

  (when (eq system-type 'darwin)
    (defun my/copy-from-osx ()
      "Handle copy/paste intelligently on osx."
      (let ((pbpaste (purecopy "/usr/bin/pbpaste")))
	(if (and (eq system-type 'darwin)
		 (file-exists-p pbpaste))
	      (let ((tramp-mode nil)
		    (default-directory "~"))
		(shell-command-to-string pbpaste)))))

    (defun my/paste-to-osx (text &optional _push)
      (let ((process-connection-type nil))
	(let ((proc (start-process "pbcopy" "*Messages*" "pbcopy")))
	  (process-send-string proc text)
	  (process-send-eof proc))))

    (unless (display-graphic-p)
      (setq interprogram-cut-function 'my/paste-to-osx
	    interprogram-paste-function 'my/copy-from-osx)))
